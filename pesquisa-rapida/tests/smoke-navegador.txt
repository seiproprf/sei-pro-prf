async page => {
  const bundle = __BUNDLE__;
  const erros = [];
  page.on('pageerror', e => erros.push(e.message));
  const config = `window.__ouvintes=[];window.chrome={storage:{sync:{get:async()=>({})},onChanged:{addListener:f=>__ouvintes.push(f),removeListener:f=>{__ouvintes=__ouvintes.filter(x=>x!==f)}}}}; window.__configurar=valor=>{for(const f of __ouvintes)f({dataValues:{newValue:JSON.stringify([{configGeral:[{name:'filtrarpaginapelapesquisarapida',value:valor}]}])}},'sync');};`;

  await page.addScriptTag({content:config});await page.addScriptTag({content:bundle});

  if(await page.locator('#p1').isHidden())throw Error('tooltip não casou');

  await page.getByRole('textbox').fill('12345');
  await page.waitForFunction(()=>getComputedStyle(document.querySelector('#p2')).display==='none');

  if(await page.locator('#p2').isHidden())throw Error('desligar não restaurou');
